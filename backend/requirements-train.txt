# Training-only dependencies for scripts/train_lstm.py.
#
# Kept separate from backend/requirements.txt so Railway's production deploy
# doesn't pull ~700MB of TensorFlow into the API container.
#
# Used by: .github/workflows/retrain-lstm.yml (weekly cron, runs in clean
# ubuntu-latest container — conflict-free by construction).
#
# Install locally (recommended: isolated venv):
#   cd backend
#   python -m venv .venv-train
#   .venv-train\Scripts\activate          # Windows
#   source .venv-train/bin/activate       # macOS/Linux
#   pip install -r requirements-train.txt

# TensorFlow 2.19 drives the numeric-stack versions. Pinning TF first and
# letting pip resolve the rest keeps numpy / protobuf / ml_dtypes / tensorboard
# all consistent with each other. TF 2.19 uses ml_dtypes>=0.5.1 — compatible
# with the onnx 1.21 in backend/requirements.txt (TF 2.16 would clash).
tensorflow-cpu==2.19.0

# Legacy Keras 2 for TF.js-compatible LayersModel export (TF.js 4.x can't parse
# Keras 3 JSON). Must match TF minor version.
tf-keras==2.19.0

# Market data. Yahoo changed their auth/crumb flow several times in 2024-2025;
# pin a recent version that handles it correctly.
yfinance>=0.2.50

# Data / ML utilities (upper bounds left open — TF 2.19's install drives the
# numpy / scipy versions; pinning here would only create resolver conflicts).
pandas>=2.2
scikit-learn>=1.4

# Parquet engine for the price-download cache (backend/scripts/.price_cache.parquet)
pyarrow>=15.0
